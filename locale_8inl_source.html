38  void set(const std::locale& value)
39  {
40 #if QX_STRING_USE_CRT_LOCALE
41  // On Windows, libc++ queries and switches the C locale for each character,
42  // causing contention on the CRT locale lock. The default named locale can
43  // instead use explicit CRT locale conversions without changing thread locale.
44  // These convert single bytes, not UTF-8 sequences; behavior for every CRT
45  // locale has not been verified. Explicit, custom and classic locales, or
46  // failure to create the CRT locale, fall back to the C++ facet.
47  const std::string sName = value.name();
48  crt_locale_ptr crtValue;
49  if (sName != "*" && sName != "C")
50  crtValue.reset(_create_locale(LC_CTYPE, sName.c_str()));
51 #endif
