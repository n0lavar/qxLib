93 constexpr std::pair<string_view, string_view> split_assert_arguments(string_view svCondition) noexcept
94 {
95  const size_t nArgsBegin = svCondition.find(QXT('('));
96  if (nArgsBegin == string_view::npos)
97  return {};
98 
99  i32 nDepth = 0;
100  size_t nCommaPos = string_view::npos;
101  size_t nArgsEnd = string_view::npos;
102 
103  for (size_t nPos = nArgsBegin + 1; nPos < svCondition.size(); ++nPos)
104  {
105  switch (svCondition[nPos])
106  {
107  case QXT('('):
108  case QXT('['):
109  case QXT('{'):
110  ++nDepth;
111  break;
112 
113  case QXT(')'):
114  if (nDepth == 0)
115  {
116  nArgsEnd = nPos;
117  nPos = svCondition.size();
118  }
119  else
120  --nDepth;
121  break;
122 
123  case QXT(']'):
124  case QXT('}'):
125  --nDepth;
126  break;
127 
128  case QXT(','):
129  if (nDepth == 0 && nCommaPos == string_view::npos)
130  nCommaPos = nPos;
131  break;
132  }
133  }
134 
135  if (nCommaPos == string_view::npos || nArgsEnd == string_view::npos)
136  return {};
137 
138  return { trim_assert_expression(svCondition.substr(nArgsBegin + 1, nCommaPos - nArgsBegin - 1)),
139  trim_assert_expression(svCondition.substr(nCommaPos + 1, nArgsEnd - nCommaPos - 1)) };
140 }
