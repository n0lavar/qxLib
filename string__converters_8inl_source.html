16 {
17  QX_PERF_SCOPE();
18 
19  if (stringView.empty())
20  {
21  out.clear();
22  return;
23  }
24 
25  out.assign(stringView.size(), L'\0');
26 
27 #if QX_STRING_USE_CRT_LOCALE
28  if (_locale_t crtLocale = &locale == &get_locale() ? details::get_locale_state().crtLocale.get() : nullptr)

30  for (size_t i = 0; i < stringView.size(); ++i)
31  {
32  wchar_t ch = static_cast<wchar_t>(WEOF);
33  if (static_cast<int>(stringView[i]) != EOF && _mbtowc_l(&ch, stringView.data() + i, 1, crtLocale) < 0)

35 
36  out[i] = ch;
37  }
38 
39  return;

41 #endif
42

44  &locale == &get_locale() ? get_locale_facet() : std::use_facet<std::ctype<wchar_t>>(locale);

46 }

159 {
160  QX_PERF_SCOPE();
161 
162 #if QX_CONF_CHAR == QX_CHAR_T_CHAR
163  out = utf8;
164 #elif QX_CONF_CHAR == QX_CHAR_T_WCHAR_T
165  #if QX_WIN
166 
167  // much faster on windows
168  // Windows accepts only int-sized input; keep the empty-result failure policy.
169  if (!std::in_range<int>(utf8.size()))

172  return;

179 
180  #else
181 
182  QX_PUSH_SUPPRESS_ALL_WARNINGS();
183  std::wstring_convert<std::codecvt_utf8_utf16<wchar_t>> converter;
184  out = converter.from_bytes(utf8.data(), utf8.data() + utf8.size());

186
