45 /**
46 
47  @class rtti_root
48  @brief Polymorphic base for qx class identity and inheritance queries.
49  @details Derive an RTTI hierarchy from this type and use QX_RTTI_CLASS in each
50  derived class to supply its identity and extend the inheritance chain.
51  This system does not require compiler RTTI.
52 
53  Static queries describe the type declaring them; virtual queries
54  describe the object's most-derived registered type. is<T>() checks
55  exact identity, while is_derived_from<T>() also accepts base types.
56 
57  The naming strategy affects class names, not class identifiers.
58  Names may be empty or shared by several types. The protected destructor
59  prevents deleting an object directly through an rtti_root pointer.
60  @tparam naming_strategy_t - strategy providing get_name(string_view) to transform
61  the compiler-generated type signature into a class name

63  @author Khrapov

197  @def QX_RTTI_CLASS
198  @brief Add dynamic identity and inheritance queries to a class in a qx RTTI hierarchy.
199  @param thisClass - type being registered, used to generate its identifier and name
200  @param ... - immediate base type, either rtti_root or a type using QX_RTTI_CLASS
201 **/
202 #define QX_RTTI_CLASS(thisClass, ...) \
203  \
204 public: \
205  using this_class_type = thisClass; \
206  using super_class_type = __VA_ARGS__; \
207  using base_class_type = typename super_class_type::base_class_type; \
208  using inheritance_tuple_type = \
209  qx::tuple_utils::join_t<typename super_class_type::inheritance_tuple_type, this_class_type>; \
210  \
211 public: \
212  static constexpr bool is_derived_from_id_static(const qx::class_id& id) noexcept \
213  { \
214  return id == get_class_id_static() || super_class_type::is_derived_from_id_static(id); \
215  } \
216  virtual bool is_derived_from_id(const qx::class_id& id) const noexcept override \
217  { \
218  static_assert(std::is_same_v<std::remove_cvref_t<decltype(*this)>, this_class_type>); \
219  return is_derived_from_id_static(id); \
220  } \
221  static constexpr qx::string_view get_class_name_static() noexcept \
222  { \
223  return base_class_type::naming_strategy_type::get_name(qx::type_strings<this_class_type>::get_signature()); \
224  } \
225  virtual qx::class_id get_class_id() const noexcept override \
226  { \
227  return get_class_id_static(); \
228  } \
229  static constexpr qx::class_id get_class_id_static() noexcept \
230  { \
231  constexpr qx::class_id id = qx::class_id::create<this_class_type>(); \
232  return id; \
233  } \
234  virtual qx::string_view get_class_name() const noexcept override \
