13 inline bool is_debugger_present() noexcept
14 {
15 #if QX_WIN
16  return ::IsDebuggerPresent() != 0;
17 #elif QX_LINUX
18  std::FILE* pStatus = std::fopen("/proc/thread-self/status", "r");
19  if (!pStatus)
20  return false;
21 
22  char buffer[512];
23  int nTracerPid = 0;
24  while (std::fgets(buffer, sizeof(buffer), pStatus))
25  {
26  if (std::sscanf(buffer, "TracerPid: %d", &nTracerPid) == 1)
27  break;
28  }
29  std::fclose(pStatus);
30  return nTracerPid != 0;
31 #elif QX_MACOS && (QX_WITH_DEBUG_INFO)
32  int mib[] = { CTL_KERN, KERN_PROC, KERN_PROC_PID, getpid() };
33  kinfo_proc info {};
34  size_t nSize = sizeof(info);
35  return sysctl(mib, 4, &info, &nSize, nullptr, 0) == 0 && (info.kp_proc.p_flag & P_TRACED) != 0;
36 #else
37  return false;
38 #endif
39 }
