220 inline flags<logger::message_necessity_type> logger::get_message_necessity_type(
221  const category& category,
222  verbosity eVerbosity,
223  std::thread::id threadId,
224  std::chrono::system_clock::time_point messageTime,
225  string_view svFile,
226  string_view svFunction,
227  int nLine) const noexcept
228 {
229  flags<message_necessity_type> eMessageNecessity;
230  {
231  std::shared_lock _(m_StreamsMutex);
232  const bool bSomeStreamRequires = std::ranges::any_of(
233  m_Streams,
234  [&category, eVerbosity, threadId, messageTime, svFile, svFunction, nLine](const auto& stream)
235  {
236  return stream->log_unconditionally_required(
237  category,
238  eVerbosity,
239  threadId,
240  messageTime,
241  svFile,
242  svFunction,
243  nLine);
244  });
245  if (bSomeStreamRequires)
246  eMessageNecessity |= message_necessity_type::one_of_streams_requires;
247  }
248 
249  {
250  std::shared_lock _(m_RegisteredCategoriesMutex);
251  if (auto itRegisteredCategory = m_RegisteredCategories.find(category.get_name());
252  itRegisteredCategory != m_RegisteredCategories.end())
253  {
254  const category_data& data = itRegisteredCategory->second;
255  if (eVerbosity >= data.eRuntimeVerbosity)
256  eMessageNecessity |= message_necessity_type::category_verbosity;
257  }
258  }
259 
260  if (eVerbosity >= verbosity::log)
261  eMessageNecessity |= message_necessity_type::default_verbosity;
262 
263  return eMessageNecessity;
264 }
